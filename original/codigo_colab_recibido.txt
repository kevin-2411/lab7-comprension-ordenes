def transcribir(audio_path, modelo_whisper=None):

    modelo = (
        modelo_whisper
        if modelo_whisper is not None
        else whisper_model
    )

    segments, info = modelo.transcribe(
        audio_path,
        language="es",
        beam_size=5
    )

    texto = " ".join(
        segmento.text.strip()
        for segmento in segments
    )

    return texto.strip()

def cargar_llm(model_id):

    print("\nCargando modelo:")
    print(model_id)

    tokenizer = AutoTokenizer.from_pretrained(
        model_id,
        trust_remote_code=True
    )

    model = AutoModelForCausalLM.from_pretrained(
        model_id,
        torch_dtype=torch.float32,
        low_cpu_mem_usage=True,
        trust_remote_code=True
    )

    model.eval()

    print("✅ Modelo cargado.")

    return tokenizer, model


def generar_llm(
    orden,
    tokenizer,
    model,
    prompt_sistema=PROMPT_SISTEMA,
    max_new_tokens=350
):

    mensajes = [
        {
            "role": "system",
            "content": prompt_sistema
        },
        {
            "role": "user",
            "content": f"Orden: {orden}"
        }
    ]

    prompt = tokenizer.apply_chat_template(
        mensajes,
        tokenize=False,
        add_generation_prompt=True
    )

    inputs = tokenizer(
        prompt,
        return_tensors="pt"
    )

    inicio = time.time()

    with torch.no_grad():

        outputs = model.generate(
            **inputs,
            max_new_tokens=max_new_tokens,
            do_sample=False,
            pad_token_id=tokenizer.eos_token_id
        )

    latencia = time.time() - inicio

    tokens_nuevos = outputs[0][
        inputs["input_ids"].shape[1]:
    ]

    respuesta = tokenizer.decode(
        tokens_nuevos,
        skip_special_tokens=True
    )

    return respuesta.strip(), latencia

# 9. EXTRAER JSON
# ============================================================

def extraer_json_de_texto(texto):

    texto = texto.strip()

    texto = texto.replace("```json", "")
    texto = texto.replace("```JSON", "")
    texto = texto.replace("```", "")

    inicio = texto.find("{")
    fin = texto.rfind("}")

    if inicio == -1 or fin == -1:
        return None

    fragmento = texto[inicio:fin + 1]

    try:
        return json.loads(fragmento)

    except json.JSONDecodeError:
        return None


def es_json_valido(texto):

    return extraer_json_de_texto(texto) is not None

# 10. RESPUESTA DEL ROBOT
# ============================================================

def generar_respuesta_robot(data):

    if data is None:
        return "No pude interpretar correctamente la orden."

    acciones = data.get("acciones", [])

    if not acciones:
        return "No entendí ninguna orden."

    partes = []

    for acc in acciones:

        accion = acc.get("accion", "")

        objetos = acc.get("objetos", [])
        personas = acc.get("personas", [])
        lugares = acc.get("lugares", [])

        elementos = []

        if objetos:

            objeto = objetos[0]

            nombre = objeto.get("nombre", "")

            adjetivos = " ".join(
                objeto.get("adjetivos", [])
            )

            texto_objeto = (
                f"{nombre} {adjetivos}"
            ).strip()

            elementos.append(texto_objeto)

        if lugares:

            lugar = lugares[0].get(
                "nombre",
                ""
            )

            elementos.append(
                f"a {lugar}"
            )

        if personas:

            persona = personas[0].get(
                "nombre",
                ""
            )

            elementos.append(
                f"a {persona}"
            )

        contenido = " ".join(elementos)

        partes.append(
            f"Voy a {accion} {contenido}".strip()
        )

    return " y ".join(partes)

def sintetizar_piper(
    texto,
    salida="respuesta_robot.wav"
):

    comando = [
        "piper",
        "--model",
        PIPER_MODEL,
        "--output_file",
        salida
    ]

    proceso = subprocess.run(
        comando,
        input=texto.encode("utf-8"),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE
    )

    if proceso.returncode != 0:

        print(
            "Error Piper:",
            proceso.stderr.decode()
        )

        return None

    return salida

def lematizar_verbo(verbo):

    doc = nlp(verbo)

    for token in doc:

        if token.pos_ in [
            "VERB",
            "AUX"
        ]:

            return token.lemma_.lower()

    return verbo.lower()

def extraer_categorias(data):

    categorias = {
        "acciones": set(),
        "objetos": set(),
        "personas": set(),
        "adjetivos": set(),
        "lugares": set()
    }

    if data is None:
        return categorias

    for accion in data.get(
        "acciones",
        []
    ):

        nombre_accion = accion.get(
            "accion",
            ""
        )

        if nombre_accion:

            categorias["acciones"].add(
                nombre_accion.lower()
            )

        for obj in accion.get(
            "objetos",
            []
        ):

            nombre = obj.get(
                "nombre",
                ""
            )

            if nombre:

                categorias["objetos"].add(
                    nombre.lower()
                )

            for adj in obj.get(
                "adjetivos",
                []
            ):

                categorias[
                    "adjetivos"
                ].add(adj.lower())

        for persona in accion.get(
            "personas",
            []
        ):

            nombre = persona.get(
                "nombre",
                ""
            )

            if nombre:

                categorias[
                    "personas"
                ].add(nombre.lower())

            for adj in persona.get(
                "adjetivos",
                []
            ):

                categorias[
                    "adjetivos"
                ].add(adj.lower())

        for lugar in accion.get(
            "lugares",
            []
        ):

            nombre = lugar.get(
                "nombre",
                ""
            )

            if nombre:

                categorias[
                    "lugares"
                ].add(nombre.lower())

            for adj in lugar.get(
                "adjetivos",
                []
            ):

                categorias[
                    "adjetivos"
                ].add(adj.lower())

    return categorias


def f1_conjuntos(real, pred):

    real = set(real)
    pred = set(pred)

    if not real and not pred:
        return 1.0

    if not real or not pred:
        return 0.0

    tp = len(
        real.intersection(pred)
    )

    precision = tp / len(pred)
    recall = tp / len(real)

    if precision + recall == 0:
        return 0.0

    return (
        2
        * precision
        * recall
        / (precision + recall)
    )


def calcular_f1_json(
    real_json,
    pred_json
):

    real = extraer_categorias(
        real_json
    )

    pred = extraer_categorias(
        pred_json
    )

    resultados = {}

    for categoria in real:

        resultados[categoria] = (
            f1_conjuntos(
                real[categoria],
                pred[categoria]
            )
        )

    resultados[
        "F1_promedio"
    ] = np.mean(
        list(resultados.values())
    )

    return resultados
